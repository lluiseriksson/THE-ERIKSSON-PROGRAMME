/-- The embedding `Q`: a real Euclidean vector dressed by `√w`, read as a
complex boundary vector.  The unitary of the charter, in its elementary
(fallback-primary) form. -/
noncomputable def qEmbed (w : (Fin L → Fin 2) → ℝ) (u : (Fin L → Fin 2) → ℝ) :
    (Fin L → Fin 2) → ℂ :=
  fun σ => ((Real.sqrt (w σ) * u σ : ℝ) : ℂ)

/-- **A1 — the isometry.**  The site form (weight `1/w`) of two dressed real
vectors is the plain Euclidean pairing.  Per-σ: `√w·u · √w·v / w = u·v`. -/
theorem siteForm_qEmbed (w : (Fin L → Fin 2) → ℝ) (hw : ∀ σ, 0 < w σ)
    (u v : (Fin L → Fin 2) → ℝ) :
    siteForm w (qEmbed w u) (qEmbed w v) = ((∑ σ, u σ * v σ : ℝ) : ℂ) := by
  unfold siteForm qEmbed
  push_cast
  refine Eq.trans (Finset.sum_congr rfl fun σ _ => ?_) ?_
  · -- conj of a real coercion is itself; then √w·√w = w cancels 1/w
    -- traps: starRingEnd on ofReal -> Complex.conj_ofReal;
    -- Real.mul_self_sqrt needs (hw σ).le
    have hs : Real.sqrt (w σ) * Real.sqrt (w σ) = w σ :=
      Real.mul_self_sqrt (hw σ).le
    have hσ : (w σ : ℝ) ≠ 0 := (hw σ).ne'
    -- goal after push_cast should be an ofReal equation; close in ℝ:
    -- (√w u)·(√w v)·(1/w) = u·v
    rw [Complex.conj_ofReal]  -- CHECK: may already be discharged by push_cast
    push_cast
    rw [show ((1 : ℂ) / (w σ : ℂ)) = ((1 / w σ : ℝ) : ℂ) by push_cast; ring]
    -- fall back to field_simp in ℝ through ofReal injectivity if the
    -- complex route fights:
    have : Real.sqrt (w σ) * u σ * (Real.sqrt (w σ) * v σ) * (1 / w σ)
        = u σ * v σ := by
      field_simp
      nlinarith [hs]
    exact_mod_cast congrArg (Complex.ofReal) this
  · norm_cast

/-- **A2 — the conjugation.**  `T ∘ Q = Q ∘ S` pointwise: the forced
operator on a dressed vector is the dressing of the symmetrised kernel's
action.  LHS σ = `w σ · Σ K σ τ · √(w τ)·u τ`; RHS σ = `√(w σ) · Σ
√(w σ)·K σ τ·√(w τ) · u τ`; equal by `√w·√w = w` pulled out of the sum. -/
theorem transferOp_qEmbed (w : (Fin L → Fin 2) → ℝ) (hw : ∀ σ, 0 < w σ)
    (β : ℝ) (u : (Fin L → Fin 2) → ℝ) :
    transferOp w β (qEmbed w u)
      = qEmbed w (fun σ => ∑ τ, symWeighted w β σ τ * u τ) := by
  funext σ
  unfold transferOp qEmbed symWeighted
  have hs : Real.sqrt (w σ) * Real.sqrt (w σ) = w σ :=
    Real.mul_self_sqrt (hw σ).le
  push_cast
  rw [Finset.mul_sum, Finset.mul_sum]
  refine Finset.sum_congr rfl fun τ _ => ?_
  push_cast
  ring_nf
  -- residue should close by rewriting w σ = √w σ · √w σ (hs) then ring;
  -- if ring_nf normalises past it, use: rw [← hs]; ring
  rw [← hs]; ring

/-- **A3 — the iterate transport**, in `act` form (seams directly into
`gibbsPathSum_eq_iterate`).  Induction on A2. -/
theorem transferOp_iterate_qEmbed (w : (Fin L → Fin 2) → ℝ)
    (hw : ∀ σ, 0 < w σ) (β : ℝ) (n : ℕ) (u : (Fin L → Fin 2) → ℝ) :
    (transferOp w β)^[n] (qEmbed w u)
      = qEmbed w ((act (symWeighted w β))^[n] u) := by
  induction n with
  | zero => rfl
  | succ n ih =>
      rw [Function.iterate_succ_apply', ih, Function.iterate_succ_apply']
      -- CHECK orientation: both iterates peel the OUTER application; A2
      -- rewrites T (Q v) with v := (act S)^[n] u, and act S v σ is
      -- definitionally Σ τ, S σ τ * v τ — the A2 RHS shape.
      exact transferOp_qEmbed w hw β ((act (symWeighted w β))^[n] u)

/-- **THE ENDPOINT (shape).**  In the Dobrushin window with β ≥ 0: OS
positivity (lane citation), the Perron data and ONE m > 0 for every
extent (D-6 verbatim), and the transport identities that read the same

theorem os_reconstruction_uniform_gap (β γ : ℝ) {α : ℝ}
    (hα0 : 0 < α) (hα1 : α < 1)
    (hwin : 2 * Real.tanh |β| + 2 * Real.tanh |γ| ≤ α) (hβ : 0 ≤ β) :
    ∃ m : ℝ, 0 < m ∧ ∀ L : ℕ,
      ∃ (lam : ℝ) (Om : (Fin (L + 1) → Fin 2) → ℝ),
        0 < lam ∧ (∀ σ, 0 < Om σ) ∧
        (∀ σ, ∑ τ, tiltKernel (sliceW γ L) β lam σ τ * Om τ = Om σ) ∧
        ‖projectedTransfer (opOf (tiltKernel (sliceW γ L) β lam))
            (vacOf Om)‖ ≤ Real.exp (-m) ∧
        -- the transported reading (A2 at w := sliceW γ L):
        (∀ u, transferOp (sliceW γ L) β (qEmbed (sliceW γ L) u)
            = qEmbed (sliceW γ L)
                (fun σ => ∑ τ, symWeighted (sliceW γ L) β σ τ * u τ)) := by
  obtain ⟨m, hm, hL⟩ := dobrushin_ising_uniform_gap β γ hα0 hα1 hwin
  refine ⟨m, hm, fun L => ?_⟩
  obtain ⟨lam, Om, hlam, hOm, hfix, hnorm⟩ := hL L
  exact ⟨lam, Om, hlam, hOm, hfix, hnorm,
    fun u => transferOp_qEmbed (sliceW γ L) (sliceW_pos γ L) β u⟩
