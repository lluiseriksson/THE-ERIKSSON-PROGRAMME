\title{A Wronskian Identity for Antisymmetrized Sine Sums:\\
\large unconditional denominator positivity via circular convolution,\\
sharp monotonicity distinctions, and counterexamples to naive sine-series principles}
\author{Lluis Eriksson}
\date{23 November 2026 (v3)}

\begin{document}
\maketitle

\begin{abstract}
For real arrays $(a_m)_{m\ge 1}$ and $(b_m)_{m\ge 1}$ that are finitely supported, or
summable with $\sum_{m\ge 1} m(|a_m|+|b_m|) < \infty$, we prove the exact Wronskian identity
\[
\sum_{1 \le m < n} (a_m b_n - a_n b_m)
\bigl[(m-n)\sin((m+n)t) + (m+n)\sin((n-m)t)\bigr]
\;=\; 2\,\bigl[F_a'(t)\,F_b(t) - F_a(t)\,F_b'(t)\bigr],
\]
where $F_x(t) = \sum_{m\ge 1} x_m \sin(mt)$, with explicit Fubini justification of the
rearrangement in the infinite-series case. Applying this identity to the modified-Bessel
coefficient arrays $(A_m, B_m)$ from a companion two-dimensional lattice gauge theory
surface-positivity problem, we prove \emph{unconditionally} for every $\beta > 0$ that
$F_B(t) = -(f^{*4})'(t)/(2 I_1(\beta)^4) > 0$ on $(0,\pi)$, where $f(t)=e^{\beta\cos t}$
and $f^{*4}$ is its fourfold normalized circular convolution (Theorem~\ref{thm:denom-pos}).
We separate precisely between: (a)~strict decrease of the real-analytic ratio $F_A/F_B$ on
$(0,\pi)$ (equivalent to $W \le 0$ and $W \not\equiv 0$), (b)~strict pointwise negativity
$W(t) < 0$ on $(0,\pi)$ (strictly stronger, as shown by $F_b=\sin t$,
$F_a = \frac{1}{4}\sin 2t + \frac{1}{8}\sin 4t$ with $F_a/F_b = \cos^3 t$ and $W(\pi/2)=0$),
and (c)~local negativity near $t=\pi$ (implied by~(a) in the real-analytic Bessel family,
but strictly weaker under generic side conditions, as shown by $a=(6,10,24)$, $b=(6,5,2)$).
We also prove nonmonotonicity of $F_a/F_b$ for $a=(3,4,3)$, $b=(3,2,1)$ via two explicit
opposite signs ($W(\pi/2) = -32 < 0$ and $W(2\pi/3) = 3\sqrt{3} > 0$), refuting any
unrestricted sine-series analogue of the Biernacki--Krzy\.z principle even when $b_m > 0$
decreases, $a_m/b_m$ strictly increases on the support, and $F_b > 0$ on $(0,\pi)$. The
finite Wronskian identity (arbitrary $K$) is machine-checked in Lean~4/Mathlib
(\texttt{WronskianIdentity.lean}, release \texttt{yang-mills-archive-v1.0}, commit
\texttt{6b54eca695e96b3038ceda18276c4e28fb45983f}), and every counterexample is an exact
algebraic evaluation. The global Bessel sign conjecture remains open.
\end{abstract}

\section{The Wronskian identity}

\begin{theorem}[Wronskian identity]
\label{thm:wronskian}
Let $(a_m)_{m\ge 1}$ and $(b_m)_{m\ge 1}$ be real sequences satisfying
$\sum_{m\ge 1} m(|a_m| + |b_m|) < \infty$ (in particular, any finitely supported sequences).
Set $F_a(t) = \sum_{m\ge 1} a_m \sin(mt)$ and $F_b(t) = \sum_{m\ge 1} b_m \sin(mt)$.
Then $F_a, F_b \in C^1(\mathbb{R})$ with termwise derivatives
$F_a'(t) = \sum_{m\ge 1} m a_m \cos(mt)$ and $F_b'(t) = \sum_{m\ge 1} m b_m \cos(mt)$,
the double series below converges absolutely and uniformly on $\mathbb{R}$, and for all
$t \in \mathbb{R}$,
\begin{equation}
\label{eq:wronskian}
\sum_{1 \le m < n} (a_m b_n - a_n b_m)
\bigl[(m-n)\sin((m+n)t) + (m+n)\sin((n-m)t)\bigr]
= 2\,\bigl[F_a'(t) F_b(t) - F_a(t) F_b'(t)\bigr].
\end{equation}
\end{theorem}

\begin{proof}
Since $|a_m \sin(mt)| \le |a_m| \le m|a_m|$ and $|m a_m \cos(mt)| \le m|a_m|$, the
Weierstrass $M$-test yields uniform convergence of $F_a, F_b, F_a', F_b'$ on $\mathbb{R}$
and validates termwise differentiation. For any $m,n \ge 1$, the product-to-sum identities
give
\begin{align*}
(m-n)\sin((m+n)t) &+ (m+n)\sin((n-m)t) \\
&= 2\,\bigl[m\cos(mt)\sin(nt) - n\sin(mt)\cos(nt)\bigr] =: 2\,G(m,n).
\end{align*}
Note that $|G(m,n)| \le m+n$, $G(n,m) = -G(m,n)$, and $G(n,n) = 0$. Moreover,
\[
\sum_{m,n \ge 1} |a_m b_n|\,(m+n)
= \Bigl(\sum_{m\ge 1} m|a_m|\Bigr)\Bigl(\sum_{n\ge 1} |b_n|\Bigr)
+ \Bigl(\sum_{m\ge 1} |a_m|\Bigr)\Bigl(\sum_{n\ge 1} n|b_n|\Bigr)
< \infty,
\]
so every double sum below is absolutely convergent and may be rearranged freely by Fubini's
theorem. Using the joint antisymmetry of $(a_m b_n - a_n b_m)$ and $G(m,n)$ together with
$G(n,n)=0$,
\begin{align*}
\sum_{1\le m < n} (a_m b_n - a_n b_m)\, 2G(m,n)
&= \sum_{m,n \ge 1} (a_m b_n - a_n b_m)\, G(m,n)
 = 2 \sum_{m,n \ge 1} a_m b_n\, G(m,n) \\
&= 2\Bigl[\Bigl(\sum_{m\ge 1} m a_m \cos(mt)\Bigr)\Bigl(\sum_{n\ge 1} b_n \sin(nt)\Bigr) \\
&\qquad\quad - \Bigl(\sum_{m\ge 1} a_m \sin(mt)\Bigr)\Bigl(\sum_{n\ge 1} n b_n \cos(nt)\Bigr)\Bigr] \\
&= 2\,\bigl[F_a'(t) F_b(t) - F_a(t) F_b'(t)\bigr]. \qedhere
\end{align*}
\end{proof}

\begin{remark}
Throughout this paper we define the scaled Wronskian with the factor of $2$ matching the
right-hand side of~\eqref{eq:wronskian}:
\[
W(t) := 2\bigl[F_a'(t) F_b(t) - F_a(t) F_b'(t)\bigr].
\]
Wherever $F_b(t) \neq 0$, the quotient rule gives
$W(t) = 2\,F_b(t)^2\,\frac{d}{dt}\bigl(F_a(t)/F_b(t)\bigr)$.
Thus on any open interval $I$ where $F_b \neq 0$, pointwise strict negativity $W(t) < 0$
on $I$ is equivalent to $(F_a/F_b)'(t) < 0$ on $I$, which is \emph{strictly stronger} than
strict decrease of $F_a/F_b$ on $I$ (see Proposition~\ref{prop:strict-vs-neg}).
\end{remark}

\section{The Bessel arrays and unconditional denominator positivity}
\label{sec:surface}

We recall the modified-Bessel coefficient arrays from the companion note \cite{PhiLemma}
(AIRR record \texttt{ARR-2026-72G1SSCNNG8YDVC2}; programme context in \cite{MasterMap}).
For $\beta > 0$, let $I_\mu(\beta) = \sum_{j=0}^\infty \frac{(\beta/2)^{\mu+2j}}{j!\,\Gamma(\mu+j+1)}$
be the modified Bessel function of the first kind, set $r_k = I_{k+1}(\beta)/I_1(\beta)$ for
$k \ge 0$, and define for $k \ge 1$,
\[
A_k = r_{k-1}^2 \!\!\sum_{j \in \{k-2,\,k\},\, j \ge 0}\!\! (j+1) r_j^2,
\qquad
B_k = k\, r_{k-1}^4 = k\,\frac{I_k(\beta)^4}{I_1(\beta)^4},
\qquad
c_{mn} = A_m B_n - A_n B_m .
\]

\begin{lemma}[Non-asymptotic super-exponential decay]
\label{lem:bessel-decay}
For every $\beta > 0$ and every integer $k \ge 0$,
\[
0 < I_k(\beta) \le \frac{(\beta/2)^k}{k!}\,I_0(\beta).
\]
Consequently, for every integer $k \ge 1$,
\[
0 < r_{k-1} = \frac{I_k(\beta)}{I_1(\beta)} \le \frac{I_0(\beta)}{I_1(\beta)}\frac{(\beta/2)^k}{k!}.
\]
In particular, for each fixed $\beta > 0$,
\[
A_m = O_\beta\!\left(m^3\left[\frac{(\beta/2)^m}{m!}\right]^4\right),
\qquad
B_m = O_\beta\!\left(m\left[\frac{(\beta/2)^m}{m!}\right]^4\right)
\qquad (m \to \infty),
\]
so $\sum_{m\ge 1} m^r(A_m + B_m) < \infty$ for every $r \ge 0$ (in particular $r=1$).
Moreover, $F_A$ and $F_B$ extend to entire (hence real-analytic on $\mathbb{R}$)
$2\pi$-periodic odd functions.
\end{lemma}

\begin{proof}
Since $\binom{k+j}{j} = \frac{(k+j)!}{k!\,j!} \ge 1$ for all $k,j \ge 0$, we have
$(k+j)! \ge k!\,j!$, whence
\[
I_k(\beta) = \sum_{j=0}^\infty \frac{(\beta/2)^{k+2j}}{j!\,(k+j)!}
\le \frac{(\beta/2)^k}{k!}\sum_{j=0}^\infty \frac{(\beta/2)^{2j}}{(j!)^2}
= \frac{(\beta/2)^k}{k!}\,I_0(\beta).
\]
Positivity follows termwise for $\beta > 0$. For $m \ge 1$, $B_m = m r_{m-1}^4$ immediately
gives $B_m = O_\beta(m [(\beta/2)^m/m!]^4)$. For $A_m$ with $m \ge 2$, using
$(\beta/2)^{m-1}/(m-1)! = (2m/\beta)\,(\beta/2)^m/m!$ gives
$r_{m-2}^2 = O_\beta(m^2 [(\beta/2)^m/m!]^2)$ while
$r_m^2 = O_\beta(m^{-2} [(\beta/2)^m/m!]^2)$, so
$(m-1)r_{m-2}^2 + (m+1)r_m^2 = O_\beta(m^3 [(\beta/2)^m/m!]^2)$ and multiplying by
$r_{m-1}^2$ yields $A_m = O_\beta(m^3 [(\beta/2)^m/m!]^4)$. Since
$|\sin(mz)| \le e^{m|\operatorname{Im} z|}$ on $\mathbb{C}$ and $m^3[(\beta/2)^m/m!]^4$
decays faster than $e^{-Rm}$ for every $R > 0$, both series converge locally uniformly on
$\mathbb{C}$ to entire $2\pi$-periodic odd functions.
\end{proof}

We now prove unconditionally that $F_B(t) > 0$ on $(0,\pi)$ for every $\beta > 0$.

\begin{lemma}[Circular convolution preserves even strict decrease]
\label{lem:conv-dec}
Let $g, h \in C^1(\mathbb{R})$ be $2\pi$-periodic and even, and suppose $g'(s) < 0$ and
$h'(s) < 0$ for all $s \in (0,\pi)$. Define their normalized circular convolution
\[
(g * h)(t) := \frac{1}{2\pi}\int_{-\pi}^{\pi} g(s)\,h(t-s)\,ds.
\]
Then $g * h \in C^1(\mathbb{R})$ is $2\pi$-periodic, even, and satisfies
$(g * h)'(t) < 0$ for all $t \in (0,\pi)$.
\end{lemma}

\begin{proof}
Periodicity and evenness follow by the substitution $s \mapsto -s$. By periodicity we may
write $(g*h)(t) = \frac{1}{2\pi}\int_{-\pi}^{\pi} g(t-u)\,h(u)\,du$, differentiate under the
integral sign, and change variables back to $s = t-u$:
\[
(g*h)'(t) = \frac{1}{2\pi}\int_{-\pi}^{\pi} g'(s)\,h(t-s)\,ds
= \frac{1}{2\pi}\int_{0}^{\pi} g'(s)\,\bigl[h(t-s) - h(t+s)\bigr]\,ds,
\]
using $g'(-s) = -g'(s)$ because $g$ is even. Because $h$ is $2\pi$-periodic, even, and
strictly decreasing on $[0,\pi]$, there is a strictly increasing function
$\tilde{h} : [-1,1] \to \mathbb{R}$ such that $h(\theta) = \tilde{h}(\cos\theta)$ for all
$\theta \in \mathbb{R}$. For any $t, s \in (0,\pi)$,
\[
\cos(t-s) - \cos(t+s) = 2\sin t\,\sin s > 0,
\]
so $h(t-s) - h(t+s) = \tilde{h}(\cos(t-s)) - \tilde{h}(\cos(t+s)) > 0$. Together with
$g'(s) < 0$ on $(0,\pi)$, the integrand is strictly negative on all of $(0,\pi)$, giving
$(g*h)'(t) < 0$.
\end{proof}

\begin{theorem}[Unconditional denominator positivity]
\label{thm:denom-pos}
For every $\beta > 0$ and every $t \in (0,\pi)$,
\[
F_B(t) = \sum_{k=1}^\infty k\,\frac{I_k(\beta)^4}{I_1(\beta)^4}\,\sin(kt)
= -\frac{(f^{*4})'(t)}{2\,I_1(\beta)^4} \;>\; 0,
\]
where $f(t) := e^{\beta\cos t}$ and $f^{*4} := (f*f)*(f*f)$.
\end{theorem}

\begin{proof}
By the Jacobi--Anger expansion \cite[10.35.2]{DLMF},
$f(t) = e^{\beta\cos t} = \sum_{k\in\mathbb{Z}} I_k(\beta) e^{ikt}$ with
$I_{-k}(\beta) = I_k(\beta)$, converging uniformly on $\mathbb{R}$. Under normalized
circular convolution, Fourier coefficients multiply, so the $k$-th Fourier coefficient of
$f^{*4}$ is $I_k(\beta)^4$:
\[
f^{*4}(t) = I_0(\beta)^4 + 2\sum_{k=1}^\infty I_k(\beta)^4 \cos(kt).
\]
By Lemma~\ref{lem:bessel-decay}, termwise differentiation is justified uniformly on
$\mathbb{R}$ and yields
\[
-(f^{*4})'(t) = 2\sum_{k=1}^\infty k\,I_k(\beta)^4\sin(kt) = 2\,I_1(\beta)^4\,F_B(t).
\]
For $\beta > 0$, $f \in C^\infty(\mathbb{R})$ is $2\pi$-periodic, even, and satisfies
$f'(t) = -\beta\sin t\,e^{\beta\cos t} < 0$ on $(0,\pi)$. Applying
Lemma~\ref{lem:conv-dec} to $(f,f)$ yields $f^{*2}$ even, $2\pi$-periodic, $C^\infty$,
and strictly decreasing on $(0,\pi)$; applying Lemma~\ref{lem:conv-dec} once more to
$(f^{*2}, f^{*2})$ gives $(f^{*4})'(t) < 0$ on $(0,\pi)$. Dividing by
$2 I_1(\beta)^4 > 0$ gives $F_B(t) > 0$ on $(0,\pi)$.
\end{proof}

\section{Sharp monotonicity distinctions and the global conjecture}
\label{sec:conjecture}

Let $W(t) := 2[F_A'(t)F_B(t) - F_A(t)F_B'(t)]$ be the Wronskian of Theorem~\ref{thm:wronskian}
at $(a,b)=(A,B)$. In the companion note \cite{PhiLemma}, the surface-positivity problem
asks for $W(t) < 0$ in a punctured left neighborhood of $t = \pi$, and proves that the
coefficient ratio $\varphi_m = A_m/B_m$ is strictly increasing ($c_{mn} < 0$ for all
$m < n$). By Theorem~\ref{thm:denom-pos}, $F_A/F_B$ is real-analytic and well-defined on
all of $(0,\pi)$, with $(F_A/F_B)' = W/(2F_B^2)$.

We clarify the exact implication structure among three candidate sign statements:
\begin{itemize}
  \item[(b)] \emph{Pointwise global negativity:} $W(t) < 0$ for all $t \in (0,\pi)$.
  \item[(a)] \emph{Weak global negativity / strict decrease:} $W(t) \le 0$ for all
  $t \in (0,\pi)$ and $W \not\equiv 0$ (equivalently, for real-analytic $F_A, F_B$ with
  $F_B > 0$, $F_A/F_B$ is strictly decreasing on $(0,\pi)$).
  \item[(c)] \emph{Near-$\pi$ local negativity:} there exists $\delta > 0$ such that
  $W(t) < 0$ for all $t \in (\pi - \delta, \pi)$.
\end{itemize}
Clearly $\text{(b)} \implies \text{(a)}$. Moreover, in the Bessel setting (where $W$ is
entire by Lemma~\ref{lem:bessel-decay}, hence analytic across $t = \pi$), if~(a) holds
then $W$ cannot vanish identically and its real zeros near $\pi$ are isolated, so
$\text{(a)} \implies \text{(c)}$ as well. Thus within the Bessel family we have the forward
chain $\text{(b)} \implies \text{(a)} \implies \text{(c)}$. Under generic finite-support
hypotheses ($b_m > 0$ decreasing, $a_m/b_m$ strictly increasing, $F_b > 0$ on $(0,\pi)$),
neither reverse implication holds: Proposition~\ref{prop:strict-vs-neg} shows
$\text{(a)} \not\Rightarrow \text{(b)}$, and Proposition~\ref{prop:local-global} shows
$\text{(c)} \not\Rightarrow \text{(a)}$. (These finite examples rule out generic reverse
implications, not Bessel-specific ones.)

\begin{proposition}[Strict decrease allows isolated zeros of $W$]
\label{prop:strict-vs-neg}
For $F_b(t) = \sin t$ (positive on $(0,\pi)$) and
$F_a(t) = \tfrac{1}{4}\sin(2t) + \tfrac{1}{8}\sin(4t)$, we have
\[
\frac{F_a(t)}{F_b(t)} = \cos^3 t,
\]
which is strictly decreasing on $(0,\pi)$, yet
$W(t) = 2\sin^2 t \cdot (-3\cos^2 t \sin t) = -6\sin^3 t\cos^2 t$ vanishes at $t = \pi/2$.
In general, for real-analytic $F_A, F_B$ with $F_B > 0$ on $(0,\pi)$, strict decrease of
$F_A/F_B$ on $(0,\pi)$ is equivalent to $W(t) \le 0$ on $(0,\pi)$ with $W \not\equiv 0$,
whereas $W(t) < 0$ on all of $(0,\pi)$ is the strictly stronger assertion that
$(F_A/F_B)'(t) < 0$ everywhere on $(0,\pi)$.
\end{proposition}

\begin{proof}
Using $\sin(2t) = 2\sin t\cos t$ and $\sin(4t) = 4\sin t\cos t(2\cos^2 t - 1)$,
\[
\frac{F_a(t)}{\sin t}
= \tfrac{1}{2}\cos t + \tfrac{1}{2}\cos t(2\cos^2 t - 1) = \cos^3 t.
\]
Hence $(F_a/F_b)'(t) = -3\cos^2 t\sin t \le 0$ with unique zero at $t = \pi/2$ in $(0,\pi)$,
so $F_a/F_b$ is strictly decreasing while $W(\pi/2)=0$. For real-analytic $F_A/F_B$ on the
connected interval $(0,\pi)$, if $W \le 0$ and $W \not\equiv 0$ then the zeros of
$(F_A/F_B)'$ are isolated, so $F_A/F_B$ is strictly decreasing; conversely, strict decrease
forces $(F_A/F_B)' \le 0$ and $(F_A/F_B)' \not\equiv 0$.
\end{proof}

\begin{conjecture}[Global Wronskian negativity for the Bessel arrays]
\label{conj:main}
For every $\beta > 0$:
\begin{enumerate}
  \item[(a)] (\emph{Weak/strict-decrease form}) $W(t) \le 0$ for all $t \in (0,\pi)$ and
  $W \not\equiv 0$ (equivalently, by Theorem~\ref{thm:denom-pos} and real-analyticity,
  $F_A/F_B$ is strictly decreasing on $(0,\pi)$).
  \item[(b)] (\emph{Strong/pointwise form}) $W(t) < 0$ for all $t \in (0,\pi)$
  (equivalently, $(F_A/F_B)'(t) < 0$ everywhere on $(0,\pi)$).
\end{enumerate}
\end{conjecture}

\section{Counterexamples to naive sine-series monotonicity and local-to-global lifting}
\label{sec:counter}

For power series with positive coefficients converging on a common interval $(0,R)$ with
nonvanishing denominator, the Biernacki--Krzy\.z principle \cite{BK} asserts that
monotonicity of $a_m/b_m$ transfers to monotonicity of $\sum a_m x^m / \sum b_m x^m$.
No unrestricted analogue holds for sine series in either direction of monotonicity, even
on finite supports where $b_m > 0$ decreases, $a_m/b_m$ strictly increases on $\{1,\dots,K\}$,
and $F_b > 0$ on $(0,\pi)$.

\begin{proposition}[Counterexample to unrestricted sine-series Biernacki--Krzy\.z]
\label{prop:counter}
Let $a = (3, 4, 3)$ and $b = (3, 2, 1)$ on indices $m \in \{1,2,3\}$ (with $a_m = b_m = 0$
for $m \ge 4$). On the support $\{1,2,3\}$, $b_m$ is strictly decreasing and the coefficient
ratio $a_m/b_m = (1,2,3)$ is strictly increasing; moreover
\[
F_b(t) = 3\sin t + 2\sin 2t + \sin 3t
= 2\sin t\,(2\cos^2 t + 2\cos t + 1) > 0 \qquad \forall t \in (0,\pi),
\]
since $2y^2 + 2y + 1 = 2(y + \tfrac{1}{2})^2 + \tfrac{1}{2} > 0$. Nevertheless,
\[
W(\pi/2) = -32 < 0
\qquad\text{and}\qquad
W(2\pi/3) = 3\sqrt{3} > 0,
\]
so $F_a/F_b$ is strictly decreasing at $t = \pi/2$ and strictly increasing at $t = 2\pi/3$,
hence non-monotone on $(0,\pi)$.
\end{proposition}

\begin{proof}
At $t = \pi/2$: $\sin t = 1$, $\sin 2t = 0$, $\sin 3t = -1$, $\cos t = 0$, $\cos 2t = -1$,
$\cos 3t = 0$, so $F_a(\pi/2) = 3 - 3 = 0$, $F_b(\pi/2) = 3 - 1 = 2$,
$F_a'(\pi/2) = -8$, and $F_b'(\pi/2) = -4$, giving
$W(\pi/2) = 2[(-8)\cdot 2 - 0\cdot(-4)] = -32 < 0$.
At $t = 2\pi/3$: $\sin t = \tfrac{\sqrt{3}}{2}$, $\sin 2t = -\tfrac{\sqrt{3}}{2}$,
$\sin 3t = 0$, $\cos t = -\tfrac{1}{2}$, $\cos 2t = -\tfrac{1}{2}$, $\cos 3t = 1$, so
$F_a = (3-4)\tfrac{\sqrt{3}}{2} = -\tfrac{\sqrt{3}}{2}$,
$F_b = (3-2)\tfrac{\sqrt{3}}{2} = \tfrac{\sqrt{3}}{2}$,
$F_a' = -\tfrac{3}{2} - 4 + 9 = \tfrac{7}{2}$, and
$F_b' = -\tfrac{3}{2} - 2 + 3 = -\tfrac{1}{2}$, giving
$F_a' F_b - F_a F_b' = \tfrac{3\sqrt{3}}{2}$ and $W(2\pi/3) = 3\sqrt{3} > 0$.
(Mere coefficient decrease does not in general force sine positivity, e.g.\
$\sin t + \sin 2t < 0$ near $\pi$; see \cite{AskeySteinig} for Vietoris-type criteria,
which is why we verify $F_b > 0$ directly by factorization.)
\end{proof}

\begin{proposition}[Near-$\pi$ negativity does not force global negativity]
\label{prop:local-global}
Let $a = (6, 10, 24)$ and $b = (6, 5, 2)$ on indices $m \in \{1,2,3\}$ (zero for $m \ge 4$).
Then on $\{1,2,3\}$, $b_m$ is strictly decreasing, $a_m/b_m = (1, 2, 12)$ is strictly
increasing, and
\[
F_b(t) = 6\sin t + 5\sin 2t + 2\sin 3t
= \sin t\,(8\cos^2 t + 10\cos t + 4) > 0 \qquad \forall t \in (0,\pi),
\]
since $10^2 - 4\cdot 8\cdot 4 = -28 < 0$. Near $t = \pi$, setting $t = \pi - \varepsilon$
yields $W(\pi - \varepsilon) = -8\varepsilon^3 + O(\varepsilon^5) < 0$ for all
sufficiently small $\varepsilon > 0$. Nevertheless, at $t = 2\pi/3$,
$W(2\pi/3) = 51\sqrt{3} > 0$. Thus any proof of Conjecture~\ref{conj:main} from near-$\pi$
negativity must invoke Bessel-specific structure beyond these generic side conditions.
\end{proposition}

\begin{proof}
Using $\sin 2t = 2\sin t\cos t$ and $\sin 3t = \sin t(4\cos^2 t - 1)$ gives
$F_b(t)/\sin t = 6 + 10\cos t + 2(4\cos^2 t - 1) = 8\cos^2 t + 10\cos t + 4 > 0$.
Similarly $F_a(t)/\sin t = 96\cos^2 t + 20\cos t - 18$. Writing $y = \cos t \in (-1,1)$,
the quotient $q(y) = F_a/F_b = (96y^2 + 20y - 18)/(8y^2 + 10y + 4)$ has derivative
\[
q'(y) = \frac{2(400 y^2 + 528 y + 130)}{(8y^2 + 10y + 4)^2}
= \frac{4(200 y^2 + 264 y + 65)}{(8y^2 + 10y + 4)^2}.
\]
Since $dy/dt = -\sin t < 0$, the sign of $W(t) = 2 F_b(t)^2 q'(\cos t)(-\sin t)$ is
opposite to the sign of $p(y) := 200 y^2 + 264 y + 65$. At $y = -1$ ($t = \pi$),
$p(-1) = 1 > 0$, confirming $W(\pi-\varepsilon) = -8\varepsilon^3 + O(\varepsilon^5) < 0$
for small $\varepsilon > 0$. At $t = 2\pi/3$ ($y = -1/2$), $p(-1/2) = 50 - 132 + 65 = -17 < 0$,
and $F_b(2\pi/3) = (\sqrt{3}/2)(2-5+4) = \sqrt{3}/2$, giving
$W(2\pi/3) = 2\cdot\frac{3}{4}\cdot\frac{4(-17)}{1^2}\cdot\bigl(-\frac{\sqrt{3}}{2}\bigr)
= 51\sqrt{3} > 0$.
\end{proof}

\begin{remark}[Structural moral and empirical numerical-precision heuristic]
\label{rem:moral}
Propositions~\ref{prop:counter} and \ref{prop:local-global} show that coefficient-ratio
monotonicity and denominator positivity alone cannot certify the sign of $W$, nor lift
local near-$\pi$ negativity to $(0,\pi)$. As an empirical numerical diagnostic (archived
in \texttt{docs/BF2-ATTACK-NOTES.md}, not part of any proof), high-precision evaluations of
the Bessel arrays at large $\beta$ exhibit severe cancellation between even- and odd-index
contributions: on sampled grids, $|W(t)|$ reaches $\sim 2.9\cdot 10^{-8}$ at $\beta=8$ and
$\sim 5.0\cdot 10^{-29}$ at $\beta=32$ while individual partial sums remain of order $1$.
Accordingly, using $\ge 2.2\beta + 20$ decimal working digits is a conservative empirical
heuristic for numerical exploration, not a proven error bound. Analytically,
Theorem~\ref{thm:denom-pos} already realizes $F_B$ as an exact circle-convolution
derivative; a corresponding positive kernel representation of $-W$ (or total positivity of
$(A;B)$ against the sine kernel) remains the natural analytic route to
Conjecture~\ref{conj:main}.
\end{remark}
